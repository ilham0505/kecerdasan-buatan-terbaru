def check_answer(self, user_answer, correct_answer):
    """Cek jawaban dengan toleransi"""
    # Normalisasi
    user_ans = str(user_answer).strip().lower()
    correct_ans = str(correct_answer).strip().lower()
    
    # Direct match
    if user_ans == correct_ans:
        return True
    
    # Numeric comparison dengan toleransi
    try:
        user_num = float(user_ans)
        correct_num = float(correct_ans)
        return abs(user_num - correct_num) < 0.01
    except:
        pass
    
    return False

def update_knowledge_level(self, current_level, is_correct, 
                           response_time, question_difficulty):
    """Update knowledge menggunakan modified Elo-like"""
    K = 0.08  # Learning rate
    
    # Expected performance
    expected_level = (question_difficulty - 1) / 4.0
    expected_prob = 1 / (1 + math.exp(-10 * (current_level - expected_level)))
    
    # Actual performance
    actual = 1.0 if is_correct else 0.0
    
    # Time factor (faster = better mastery)
    time_factor = 1.0
    if is_correct:
        if response_time < 5:
            time_factor = 1.2  # Bonus untuk cepat
        elif response_time > 20:
            time_factor = 0.9  # Penalti untuk lambat
    
    # Update calculation
    delta = K * (actual - expected_prob) * time_factor
    new_level = current_level + delta
    
    return max(0.0, min(1.0, new_level))
python# adaptive_engine.py - baris 105-192
def determine_next_action(self, is_correct, emotion, knowledge_level,
                          questions_attempted, questions_correct):
    """Decision engine untuk aksi berikutnya"""
    
    accuracy = questions_correct / questions_attempted
    
    # HIGH ANXIETY - prioritas!
    if emotion in ['frustrated', 'confused', 'anxious', 'stressed']:
        if not is_correct:
            return {
                'action': 'review_material',  # ← UPDATE: Review materi
                'recommendation': 'Mari review dengan cara lebih mudah'
            }
    
    # STRUGGLING
    if accuracy < 0.4 or knowledge_level < 0.4:
        return {
            'action': 'easier_content',  # ← UPDATE: Materi lebih mudah
            'recommendation': 'Mari coba pendekatan lain'
        }
    
    # MASTERY
    if knowledge_level > 0.8 and accuracy > 0.8:
        return {
            'action': 'end_session',  # ← Selesai
            'recommendation': 'Hebat! Siap topik baru?'
        }
    
    # MODERATE
    return {
        'action': 'continue_practice',  # ← Lanjut dengan adjustment
        'recommendation': 'Bagus! Yuk lanjut!'
    }
